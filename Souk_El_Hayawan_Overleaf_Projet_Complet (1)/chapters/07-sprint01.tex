\section{UC-S01-01 --- Consulter les races d’une catégorie}
\begin{longtable}{@{}P{35mm}P{108mm}@{}}\caption{Description du cas UC-S01-01}\label{tab:s01-uc1}\\
\toprule\textbf{Rubrique} & \textbf{Description}\\\midrule\endfirsthead\toprule\textbf{Rubrique} & \textbf{Description}\\\midrule\endhead
\UCrow{Code}{UC-S01-01}
\UCrow{Acteur principal}{Visiteur}
\UCrow{Objectif}{Consulter les races actives d’une catégorie d’animaux.}
\UCrow{Préconditions}{Le catalogue persistant contient des catégories valides ; le service de consultation est disponible.}
\UCrow{Déclencheur}{Choix d’une catégorie dans l’interface.}
\UCrow{Scénario nominal}{1. Charger les catégories actives.\newline 2. Sélectionner une catégorie.\newline 3. Angular appelle l’API des races.\newline 4. Le backend vérifie la catégorie et retourne les races actives.\newline 5. L’interface affiche la liste.}
\UCrow{Alternatives}{Catégorie inconnue : message adapté ; aucune race : état vide explicite ; API indisponible : gestion d’erreur sans fausses données.}
\UCrow{Règles métier}{Une race appartient à une catégorie ; seules les catégories et races actives sont visibles.}
\UCrow{Postconditions}{Le visiteur voit les races réellement présentes dans la base.}
\UCrow{Critères}{API et interface synchronisées ; changement de catégorie actualise les races ; tests des états vide/invalide.}
\bottomrule\end{longtable}

\section{Tests attendus}
\begin{itemize}
\item Catégories actives listées ; catégories inactives absentes du résultat public.
\item Races d’une catégorie obtenues par une requête réellement exécutée.
\item Catégorie inconnue et catégorie vide traitées sans affichage trompeur.
\item Intégrité relationnelle vérifiée par contrainte de clé étrangère.
\item Données de référence sourcées et migrations reproductibles.
\item Interface Angular raccordée à la vraie API ; tests manuels et automatisés.
\end{itemize}
